--- Cascade fit (CPU, cascade_math.cpp; the GLSL consumes only the results) ---
  splits: z_i = lerp( n+(f-n)*i/4 , n*pow(f/n, i/4) , 0.80 ) for i=0..4, n=1.0, f=1500.0  (z_0=n, z_4=f).

  Light basis: upref = (|L.y|<0.99) ? (0,1,0) : (0,0,-1); r = normalize(cross(upref,L)); u = cross(L,r)   [(r,u,L) is right-handed; light looks along -L].
  cx=dot(r,C), cy=dot(u,C), cz=dot(L,C); T=2R/SHADOW_RES; cx=floor(cx/T+0.5)*T; cy=floor(cy/T+0.5)*T.

--- PBR-lite (pbr_lite.glsl; CPU twin pbr_ref.cpp; formulas FIXED and literally mirrored) ---
  pbr_shade(albedo, rough, metal, ao, N, V, L, sun_rgb, vis, sky, ground, sky_i):
   rc=clamp(rough,0.04,1); a=rc*rc; a2=a*a; NL=clamp(dot(N,L),0,1); NV=max(dot(N,V),1e-4); H=normalize(L+V); NH=clamp(dot(N,H),0,1); VH=clamp(dot(V,H),0,1);
   D=a2/(PI*sqr(NH*NH*(a2-1)+1)); Vis=0.5/max(NL*sqrt(NV*NV*(1-a2)+a2)+NV*sqrt(NL*NL*(1-a2)+a2),1e-5);
   F0=mix(vec3(0.04),albedo,metal); F=F0+(1-F0)*pow(1-VH,5);
