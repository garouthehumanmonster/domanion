    select(node, L):  // returns false if node is outside range[L]
      if !sphere_hits_aabb(cam, range[L], node.aabb) return false
      if frustum && !frustum_hits_aabb(node.aabb) return true       // culled, nothing to draw
      if L == 0 { add(node, 0); return true }
      if !sphere_hits_aabb(cam, range[L-1], node.aabb) { add(node, L); return true }
      for each child c (z-major, x-minor order): if !select(c, L-1) add(c, L-1)
      return true

  Node AABB = [x, x+size] x [minh, maxh] x [z, z+size]; minh/maxh per node precomputed from
